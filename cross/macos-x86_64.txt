# x86-64 macOS build of the unix half of the Wine build, d3d12metal.so. Wine here

#   meson setup build-wine/unix --cross-file cross/macos-x86_64.txt

ar = 'ar'
strip = 'strip'

[host_machine]
system = 'darwin'
cpu_family = 'x86_64'
cpu = 'x86_64'
endian = 'little'

[properties]

[project options]
